from machine import Pin, PWM
import time
import random

# 2 pins de direction par moteur (IN1/IN2) + 1 PWM pin (ENA/ENB).
# Les pins sont théorique, je ne les ai pas testées. Nous allons devoir les ajuster.
# Mettre invert=True si le moteur est branché à l'envers (roue tourne dans le mauvais sens).
MOTORS = {
    "FL": dict(in1=25, in2=26, en=27, invert=False),  # Roue avant droite
    "FR": dict(in1=32, in2=33, en=14, invert=False),  # Roue avant gauche
    "RL": dict(in1=36, in2=37, en=22, invert=False),  # Roue arrière droite
    "RR": dict(in1=15, in2=16, en=17, invert=False),  # Roue arrière gauche
}

FRONT_SENSOR_PIN = 34   # Pin input qui a besoin d'un pull-up externe.
BACK_SENSOR_PIN = 35
SENSOR_ACTIVE_LEVEL = 0  # E18-D80NK output est LOW quand un obstacle est détecté.

# Speeds are 0.0 - 1.0
CRUISE_SPEED = 0.60
REVERSE_SPEED = 0.50
TURN_SPEED = 0.60
MIN_DUTY = 0.35          # This indiquates the minimum duty cycle that actually moves the motors. Below this, the motors won't turn.
PWM_FREQ = 1000          # FREQUENCE IN Hz

# Timings (ms)
LOOP_MS = 20
PAUSE_MS = 150           # short stop between direction changes. Change it if it does not feel right.
REVERSE_MS = 600
TURN_MIN_MS = 500
TURN_MAX_MS = 1800       # If the rover is still blocked after turning this much time, it will try the other turn direction.
CLEAR_CONFIRM_MS = 300   # front must stay clear this long before driving off.
STUCK_WAIT_MS = 2000
MAX_TURN_ATTEMPTS = 4    # after this many failed turns the rover will go into STUCK mode and wait a bit before trying again.
START_DELAY_MS = 3000    # time to put the rover down / press CTRL-C before it starts moving. You can set it to 0 if you want to start immediately.

SENSOR_FILTER = 3        # consecutive readings needed to change state.



# DRIVERS
# =====================================================================
class Motor:
    def __init__(self, in1, in2, en, invert=False):
        self.in1 = Pin(in1, Pin.OUT, value=0)
        self.in2 = Pin(in2, Pin.OUT, value=0)
        self.pwm = PWM(Pin(en), freq=PWM_FREQ, duty_u16=0)
        self.invert = invert

    def set(self, speed):
        """speed: -1.0 (full reverse) ... 0 (stop) ... 1.0 (full forward)"""
        speed = max(-1.0, min(1.0, speed))
        if self.invert:
            speed = -speed
        if speed == 0:
            self.in1.value(0)
            self.in2.value(0)
            self.pwm.duty_u16(0)
            return
        duty = MIN_DUTY + (1.0 - MIN_DUTY) * abs(speed)
        self.in1.value(1 if speed > 0 else 0)
        self.in2.value(0 if speed > 0 else 1)
        self.pwm.duty_u16(int(duty * 65535))


class ObstacleSensor:
    """E18-D80NK with a simple filter against noisy readings."""

    def __init__(self, pin_no):
        self.pin = Pin(pin_no, Pin.IN)
        self._count = 0
        self.detected = False

    def update(self):
        if self.pin.value() == SENSOR_ACTIVE_LEVEL:
            self._count = min(self._count + 1, SENSOR_FILTER)
        else:
            self._count = max(self._count - 1, 0)
        if self._count >= SENSOR_FILTER:
            self.detected = True
        elif self._count == 0:
            self.detected = False

# ROVER
# =====================================================================
FORWARD = "FORWARD"        # No shit sherlock.
REVERSE = "REVERSE"
TURN = "TURN"
PAUSE = "PAUSE"
STUCK = "STUCK"


class Rover:
    def __init__(self):
        m = {name: Motor(**cfg) for name, cfg in MOTORS.items()}
        self.left = [m["FL"], m["RL"]]
        self.right = [m["FR"], m["RR"]]
        self.front = ObstacleSensor(FRONT_SENSOR_PIN)
        self.back = ObstacleSensor(BACK_SENSOR_PIN)

        self.mode = "auto"
        self.state = None
        self.t0 = time.ticks_ms()
        self._after = FORWARD
        self.turn_dir = 1          # If thou were not knowledgable, +1 = turn right, -1 = turn left
        self.turn_attempts = 0
        self._clear_since = None
        self.stop()

    # PUBLIC API, CAN BE USED BY THE MANUAL MODE WE WILL ADD LATER :)
    def drive(self, left, right):
        """Tank control: each side from -1.0 to +1.0, like a tank '-' """
        for mot in self.left:
            mot.set(left)
        for mot in self.right:
            mot.set(right)

    def stop(self):
        self.drive(0, 0)

    def set_mode(self, mode):
        self.stop()
        self.mode = mode
        if mode == "auto":
            self.turn_attempts = 0
            self._enter(FORWARD)

    def update(self):
        """Call every 20 ms. Non-blocking (I hope)."""
        self.front.update()
        self.back.update()
        if self.mode == "auto":
            self._auto_step()
        # elif self.mode == "manual": handled later.
        # To block forward/backward commands (so it doesn't ram into things like me, you know?).

    # ------- State machine ------- (Visual studio told me to say it like that, so I will. I don't know what it means though.)
    def _enter(self, state):
        self.state = state
        self.t0 = time.ticks_ms()
        print("state:", state)
        if state == FORWARD:
            self.turn_attempts = 0
            self.drive(CRUISE_SPEED, CRUISE_SPEED)
        elif state == REVERSE:
            self.drive(-REVERSE_SPEED, -REVERSE_SPEED)
        elif state == TURN:
            self._clear_since = None
            t = TURN_SPEED * self.turn_dir
            self.drive(t, -t)      # spin in place
        else:                      # PAUSE, STUCK
            self.stop()

    def _pause(self, then):
        self._after = then
        self._enter(PAUSE)

    def _random_turn(self):
        self.turn_dir = 1 if random.getrandbits(1) else -1
        self.turn_attempts = 0

    def _auto_step(self):
        now = time.ticks_ms()
        elapsed = time.ticks_diff(now, self.t0)
        s = self.state

        if s == FORWARD:
            if self.front.detected:
                self._random_turn()
                self._pause(TURN if self.back.detected else REVERSE)

        elif s == REVERSE:
            if self.back.detected or elapsed >= REVERSE_MS:
                self._pause(TURN)

        elif s == TURN:
            if self.front.detected:
                self._clear_since = None
                if elapsed >= TURN_MAX_MS:
                    self.turn_attempts += 1
                    if self.turn_attempts >= MAX_TURN_ATTEMPTS:
                        self._enter(STUCK)
                    else:
                        self.turn_dir = -self.turn_dir   # try the other way
                        self._enter(TURN)
            else:
                if self._clear_since is None:
                    self._clear_since = now
                elif (time.ticks_diff(now, self._clear_since) >= CLEAR_CONFIRM_MS
                      and elapsed >= TURN_MIN_MS):
                    self._pause(FORWARD)

        elif s == PAUSE:
            if elapsed >= PAUSE_MS:
                self._enter(self._after)

        elif s == STUCK:
            if elapsed >= STUCK_WAIT_MS:
                self._random_turn()
                self._pause(TURN)

# MAIN, DUH
# =====================================================================
def main():
    rover = Rover()
    print("Starting in %d ms (Ctrl-C to abort)" % START_DELAY_MS)
    time.sleep_ms(START_DELAY_MS)
    rover.set_mode("auto")
    try:
        while True:
            rover.update()
            time.sleep_ms(LOOP_MS)
    except KeyboardInterrupt:
        pass
    finally:
        rover.stop()
        print("Stopped.")

main()
